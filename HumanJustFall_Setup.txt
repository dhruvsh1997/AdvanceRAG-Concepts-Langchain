-- Paste ALL of this into the Command Bar in Roblox Studio and press Enter.
-- (View tab -> Command Bar)

local SSS = game:GetService("ServerScriptService")
local RS = game:GetService("ReplicatedStorage")
local SPS = game:GetService("StarterPlayer"):WaitForChild("StarterPlayerScripts")
local WS = game:GetService("Workspace")

local function clear(parent, name)
	local o = parent:FindFirstChild(name)
	if o then o:Destroy() end
end

---------------------------------------------------------------- remotes
clear(RS, "HJF_Remotes")
local remotes = Instance.new("Folder")
remotes.Name = "HJF_Remotes"
remotes.Parent = RS
for _, n in ipairs({ "PlayMode", "RagdollToggle", "Grab", "Release", "Message" }) do
	local r = Instance.new("RemoteEvent")
	r.Name = n
	r.Parent = remotes
end

---------------------------------------------------------------- server script
local SERVER_SRC = [==[
local Players = game:GetService("Players")
local RS = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local TeleportService = game:GetService("TeleportService")
local remotes = RS:WaitForChild("HJF_Remotes")

local REACH = 10
local FALL_SPEED = -60
local ragdolled = {}
local grabs = {}

local function setRagdoll(char, on)
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return end
	if on and not ragdolled[char] then
		ragdolled[char] = true
		hum:ChangeState(Enum.HumanoidStateType.Physics)
		for _, m in ipairs(char:GetDescendants()) do
			if m:IsA("Motor6D") and m.Part0 and m.Part1 then
				local a0 = Instance.new("Attachment")
				a0.Name = "HJF_A0"
				a0.CFrame = m.C0
				a0.Parent = m.Part0
				local a1 = Instance.new("Attachment")
				a1.Name = "HJF_A1"
				a1.CFrame = m.C1
				a1.Parent = m.Part1
				local b = Instance.new("BallSocketConstraint")
				b.Name = "HJF_Ball"
				b.Attachment0 = a0
				b.Attachment1 = a1
				b.LimitsEnabled = true
				b.UpperAngle = 70
				b.TwistLimitsEnabled = true
				b.TwistLowerAngle = -45
				b.TwistUpperAngle = 45
				b.Parent = m.Part0
				m.Enabled = false
			end
		end
	elseif (not on) and ragdolled[char] then
		ragdolled[char] = nil
		for _, d in ipairs(char:GetDescendants()) do
			if d.Name == "HJF_Ball" or d.Name == "HJF_A0" or d.Name == "HJF_A1" then
				d:Destroy()
			elseif d:IsA("Motor6D") then
				d.Enabled = true
			end
		end
		local root = char:FindFirstChild("HumanoidRootPart")
		if root then
			root.CFrame = CFrame.new(root.Position + Vector3.new(0, 3, 0))
		end
		hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	end
end

local function getHand(char, side)
	return char:FindFirstChild(side .. "Hand") or char:FindFirstChild(side .. " Arm")
end

local function startPlaying(plr)
	plr:SetAttribute("Playing", true)
	local char = plr.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.WalkSpeed = 16
		hum.JumpHeight = 7.2
		hum.JumpPower = 50
	end
end

local function onChar(plr, char)
	local hum = char:WaitForChild("Humanoid")
	local root = char:WaitForChild("HumanoidRootPart")
	if not plr:GetAttribute("Playing") then
		hum.WalkSpeed = 0
		hum.JumpHeight = 0
		hum.JumpPower = 0
	end
	local minY = 0
	local conn
	conn = RunService.Heartbeat:Connect(function()
		if not char.Parent then
			conn:Disconnect()
			ragdolled[char] = nil
			return
		end
		if hum:GetState() == Enum.HumanoidStateType.Freefall then
			minY = math.min(minY, root.AssemblyLinearVelocity.Y)
		end
	end)
	hum.StateChanged:Connect(function(_, new)
		if new == Enum.HumanoidStateType.Landed then
			if minY < FALL_SPEED and not ragdolled[char] then
				setRagdoll(char, true)
				task.delay(2.5, function()
					if char.Parent then setRagdoll(char, false) end
				end)
			end
			minY = 0
		end
	end)
end

local function onPlayer(plr)
	local join = plr:GetJoinData()
	local data = join and join.TeleportData
	if type(data) == "table" and data.AutoStart then
		plr:SetAttribute("Playing", true)
	end
	plr.CharacterAdded:Connect(function(c) onChar(plr, c) end)
	plr.CharacterRemoving:Connect(function() grabs[plr] = nil end)
	if plr.Character then onChar(plr, plr.Character) end
end
Players.PlayerAdded:Connect(onPlayer)
for _, p in ipairs(Players:GetPlayers()) do onPlayer(p) end

remotes.PlayMode.OnServerEvent:Connect(function(plr, mode)
	if plr:GetAttribute("Playing") then return end
	if mode == "Single" and game.PrivateServerId == "" and not RunService:IsStudio() then
		local ok = pcall(function()
			local code = TeleportService:ReserveServer(game.PlaceId)
			TeleportService:TeleportToPrivateServer(game.PlaceId, code, { plr }, nil, { AutoStart = true })
		end)
		if ok then return end
	end
	startPlaying(plr)
end)

remotes.RagdollToggle.OnServerEvent:Connect(function(plr)
	local char = plr.Character
	if not char or not plr:GetAttribute("Playing") then return end
	setRagdoll(char, not ragdolled[char])
end)

remotes.Grab.OnServerEvent:Connect(function(plr, side, part, pos)
	if side ~= "Left" and side ~= "Right" then return end
	if typeof(part) ~= "Instance" or not part:IsA("BasePart") then return end
	if typeof(pos) ~= "Vector3" then return end
	local char = plr.Character
	if not char or not plr:GetAttribute("Playing") then return end
	if part:IsDescendantOf(char) then return end
	local hand = getHand(char, side)
	if not hand then return end
	if (hand.Position - pos).Magnitude > REACH then return end
	grabs[plr] = grabs[plr] or {}
	if grabs[plr][side] then grabs[plr][side]:Destroy() end
	local w = Instance.new("WeldConstraint")
	w.Part0 = hand
	w.Part1 = part
	w.Parent = hand
	grabs[plr][side] = w
end)

remotes.Release.OnServerEvent:Connect(function(plr, side)
	local g = grabs[plr]
	if g and g[side] then
		g[side]:Destroy()
		g[side] = nil
	end
end)

-- finish pad
local finish = workspace:WaitForChild("FinishPad", 10)
if finish then
	finish.Touched:Connect(function(hit)
		local model = hit:FindFirstAncestorOfClass("Model")
		local plr = model and Players:GetPlayerFromCharacter(model)
		if plr and not plr:GetAttribute("Won") then
			plr:SetAttribute("Won", true)
			remotes.Message:FireClient(plr, "YOU FINISHED THE MAP!")
			task.wait(4)
			plr:LoadCharacter()
			plr:SetAttribute("Won", false)
		end
	end)
end
]==]

---------------------------------------------------------------- client script
local CLIENT_SRC = [==[
local Players = game:GetService("Players")
local RS = game:GetService("ReplicatedStorage")
local UIS = game:GetService("UserInputService")
local player = Players.LocalPlayer
local remotes = RS:WaitForChild("HJF_Remotes")

local function make(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do o[k] = v end
	o.Parent = parent
	return o
end

local gui = make("ScreenGui", { Name = "HJFGui", ResetOnSpawn = false, IgnoreGuiInset = true }, player:WaitForChild("PlayerGui"))

local menu = make("Frame", { Name = "Menu", Size = UDim2.fromScale(1, 1), BackgroundColor3 = Color3.fromRGB(14, 24, 38) }, gui)
make("TextLabel", {
	Size = UDim2.fromScale(0.8, 0.14), AnchorPoint = Vector2.new(0.5, 0), Position = UDim2.fromScale(0.5, 0.15),
	BackgroundTransparency = 1, Text = "HUMAN JUST FALL", Font = Enum.Font.GothamBlack,
	TextScaled = true, TextColor3 = Color3.fromRGB(255, 170, 60),
}, menu)
make("TextLabel", {
	Size = UDim2.fromScale(0.6, 0.04), AnchorPoint = Vector2.new(0.5, 0), Position = UDim2.fromScale(0.5, 0.30),
	BackgroundTransparency = 1, Text = "wobble. grab. climb. fall.", Font = Enum.Font.Gotham,
	TextScaled = true, TextColor3 = Color3.fromRGB(150, 180, 210),
}, menu)

local function button(text, y, callback)
	local b = make("TextButton", {
		Size = UDim2.fromOffset(300, 60), AnchorPoint = Vector2.new(0.5, 0), Position = UDim2.new(0.5, 0, y, 0),
		BackgroundColor3 = Color3.fromRGB(40, 70, 110), Text = text, Font = Enum.Font.GothamBold,
		TextSize = 26, TextColor3 = Color3.new(1, 1, 1),
	}, menu)
	make("UICorner", { CornerRadius = UDim.new(0, 12) }, b)
	b.Activated:Connect(callback)
end
button("SINGLE PLAYER", 0.45, function() remotes.PlayMode:FireServer("Single") end)
button("MULTIPLAYER", 0.58, function() remotes.PlayMode:FireServer("Multi") end)

local hint = make("TextLabel", {
	Size = UDim2.fromOffset(520, 28), Position = UDim2.new(0, 12, 1, -40), BackgroundTransparency = 1,
	Text = "Left click / Right click = grab with left / right hand     R = ragdoll on/off",
	Font = Enum.Font.GothamMedium, TextSize = 16, TextColor3 = Color3.new(1, 1, 1),
	TextXAlignment = Enum.TextXAlignment.Left, Visible = false,
}, gui)

local msg = make("TextLabel", {
	Size = UDim2.fromScale(0.8, 0.12), AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.35),
	BackgroundTransparency = 1, Text = "", Font = Enum.Font.GothamBlack, TextScaled = true,
	TextColor3 = Color3.fromRGB(120, 255, 140), Visible = false,
}, gui)
remotes.Message.OnClientEvent:Connect(function(text)
	msg.Text = text
	msg.Visible = true
	task.delay(3.5, function() msg.Visible = false end)
end)

local function refresh()
	local playing = player:GetAttribute("Playing") == true
	menu.Visible = not playing
	hint.Visible = playing
end
player:GetAttributeChangedSignal("Playing"):Connect(refresh)
refresh()

-- grabbing + ragdoll controls
local mouse = player:GetMouse()
local function setFilter(c) mouse.TargetFilter = c end
if player.Character then setFilter(player.Character) end
player.CharacterAdded:Connect(setFilter)

local function tryGrab(side)
	if player:GetAttribute("Playing") ~= true then return end
	local t = mouse.Target
	if t then remotes.Grab:FireServer(side, t, mouse.Hit.Position) end
end

UIS.InputBegan:Connect(function(input, gp)
	if gp then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		tryGrab("Left")
	elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
		tryGrab("Right")
	elseif input.KeyCode == Enum.KeyCode.R then
		remotes.RagdollToggle:FireServer()
	end
end)

UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		remotes.Release:FireServer("Left")
	elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
		remotes.Release:FireServer("Right")
	end
end)
]==]

---------------------------------------------------------------- install scripts
local function mk(class, name, parent, src)
	clear(parent, name)
	local s = Instance.new(class)
	s.Name = name
	s.Source = src
	s.Parent = parent
end
mk("Script", "HJF_Server", SSS, SERVER_SRC)
mk("LocalScript", "HJF_Client", SPS, CLIENT_SRC)

local function block(name, pos, size, color, parent)
	local p = Instance.new("Part")
	p.Name = name
	p.Anchored = true
	p.Size = size
	p.Position = pos
	p.Color = color
	p.Material = Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	p.Parent = parent or map
	return p
end
